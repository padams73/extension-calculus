\subsection*{Round 2 answers}
\begin{enumerate}[itemsep=1cm]
    \item 
    $\frac{x^3}{3}\ln(x)-\frac{x^3}{9}+c$

    \item 
    Use Camel Principle additively:
    $\int \frac{x^3+x}{x^2 +1} - \frac{x}{x^2 +1}\,dx$

    $\int \frac{x(x^2+1)}{x^2 +1}\,dx - \frac{1}{2}\int \frac{2x}{x^2 +1}\,dx$

    $\int x\,dx - \frac{1}{2}\int \frac{2x}{x^2 +1}\,dx$
    
    $\frac{x^2}{2}-\frac{1}{2}\ln(x^2 + 1)+c$

    \item
    $\ln{|\sec(x)+\tan(x)|}+c$

    \item
    $\int \frac{1}{1+\frac{\sin{x}}{\cos{x}}}\,dx$

    $\int \frac{1}{\frac{\cos{x}+\sin{x}}{\cos{x}}}\,dx$

    $\int \frac{\cos{x}}{\sin{x}+\cos{x}}\,dx$

    $\frac{1}{2}\int \frac{2\cos{x}}{\sin{x}+\cos{x}}\,dx$

    $\frac{1}{2}\int \frac{2\cos{x}+\sin{x}-\sin{x}}{\sin{x}+\cos{x}}\,dx$

    $\frac{1}{2}\int \Bigl(\frac{\cos{x}+\sin{x}}{\sin{x}+\cos{x}}+\frac{\cos{x}-\sin{x}}{\sin{x}+\cos{x}}\Bigr)\,dx$
    
    $\frac{1}{2}\int 1\,dx+\frac{1}{2}\int \frac{\cos{x}-\sin{x}}{\sin{x}+\cos{x}}\,dx$
    
    $\frac{x}{2}+\frac{1}{2}\ln{|\sin{x}+\cos{x}|}+c$

    \item
    $\frac{x\sqrt{1-x^2}}{2}+\frac{\sin^{-1}(x)}{2}+c$

    \item
    Use Camel Principle additively:

    $\int \frac{1+e^x}{1+e^x}\,dx - \int \frac{e^x}{1+e^x}\,dx$

    $\int 1\,dx - \int \frac{e^x}{1+e^x}\,dx$

    $x-\ln(1+e^x)+c$

    Or use Camel Principle multiplicatively:

    $\int \frac{e^{-x}}{e^{-x}+1}\,dx$

    $-\int \frac{-e^{-x}}{e^{-x}+1}\,dx$

    $-\ln(e^{-x}+1)+c$

\end{enumerate}

\subsection*{Finals answers}
\begin{enumerate}[itemsep=1cm]
    \item 
    $u= \sqrt{x}$
    
    $u^2 = x$
    
    $2u\,du = dx$
    
    $2\int u e^u\,du$
    
    Using Integration by Parts gives:
    
    $2e^{\sqrt{x}}(\sqrt{x}-1)+c$

    \item 
    Using a trig substitution we rewrite the integral as:
    
    $\frac{3}{4} \int \frac{\sec^2(\theta)}{9+9\tan^2(\theta)}\,d\theta$
    
    $\frac{3}{4} \int \frac{\sec^2(\theta)}{9(1+\tan^2(\theta))}\,d\theta$
    
    $\frac{3}{4} \int \frac{\sec^2(\theta)}{9(\sec^2(\theta))}\,d\theta$
    
    $\frac{1}{12} \int 1\,d\theta$
    
    $\frac{1}{12}\theta +c$
    
    $\frac{1}{12} \tan^{-1}\Bigl(\frac{2x^2}{3}\Bigr)+c$

    \item
    Integrate by parts, setting $f'=1$ and $g=\ln(2+x^2)$
    
    $\int \ln(2+x^2)\,dx = x\ln(2+x^2)-\int \frac{2x^2}{2+x^2}$

    Integrate the second part by using a trig substitution:

    $\tan(\theta)=\frac{x}{\sqrt{2}}$

    After simplification we get:

    $2\sqrt{2}\int \tan^2(\theta)\,d\theta$
    
    $2\sqrt{2}\int \sec^2(\theta)-1\,d\theta$

    $2\sqrt{2}\tan(\theta)-2\sqrt{2}\theta+c$

    Reversing substitution:

    $2x - 2\sqrt{2}\tan^{-1}\Bigl(\frac{x}{\sqrt{2}}\Bigr)+c$

    Combining with first part of original integral, the answer is:

    $x\ln(2+x^2)-2x+2\sqrt{2}\tan^{-1}\Bigl(\frac{x}{\sqrt{2}}\Bigr)+c$


\end{enumerate}
